   138→// ---------- lead picker (searchable, debounced) ----------
   139→
   140→function LeadPicker({ value, onSelect }: { value: PickerLead | null; onSelect: (l: PickerLead | null) => void }) {
   141→  const [q, setQ] = useState('')
   142→  const [results, setResults] = useState<PickerLead[]>([])
   143→  const [loading, setLoading] = useState(false)
   144→
   145→  useEffect(() => {
   146→    if (value) return
   147→    let cancelled = false
   148→    const t = setTimeout(() => {
   149→      setLoading(true)
   150→      api<{ leads: PickerLead[] }>(`/api/leads${qs({ q: q || undefined, pageSize: 10 })}`)
   151→        .then((res) => {
   152→          if (!cancelled) setResults(res.leads ?? [])
   153→        })
   154→        .catch(() => {
   155→          if (!cancelled) setResults([])
   156→        })
   157→        .finally(() => {
   158→          if (!cancelled) setLoading(false)
   159→        })
   160→    }, 350)
   161→    return () => {
   162→      cancelled = true
   163→      clearTimeout(t)
   164→    }
   165→  }, [q, value])
   166→
   167→  if (value) {
   168→    return (
   169→      <div className="flex items-center justify-between rounded-lg border border-emerald-200 bg-emerald-50 px-3 py-2">
   170→        <div className="min-w-0">
   171→          <p className="truncate text-sm font-medium text-stone-800">
   172→            {value.customerName} <span className="font-mono text-xs text-stone-500">{value.leadCode}</span>
   173→          </p>
   174→          <p className="text-xs text-stone-500">
   175→            {value.mobile} · {value.department}
   176→          </p>
   177→        </div>
   178→        <Button type="button" variant="ghost" size="sm" onClick={() => onSelect(null)}>
   179→          Change
   180→        </Button>
   181→      </div>
   182→    )
   183→  }
   184→
   185→  return (
   186→    <div className="space-y-1.5">
   187→      <Input placeholder="Search lead by name / mobile / code..." value={q} onChange={(e) => setQ(e.target.value)} />
   188→      <div className="max-h-44 overflow-y-auto rounded-lg border border-stone-200">
   189→        {loading ? (
   190→          <div className="flex items-center gap-2 px-3 py-3 text-xs text-stone-500">
   191→            <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden /> Searching...
   192→          </div>
   193→        ) : results.length === 0 ? (
   194→          <p className="px-3 py-3 text-xs text-stone-500">No leads found</p>
   195→        ) : (
   196→          results.map((l) => (
   197→            <button
   198→              key={l.id}
   199→              type="button"
   200→              onClick={() => onSelect(l)}
   201→              className="flex w-full items-center justify-between gap-2 border-b border-stone-100 px-3 py-2 text-left text-sm last:border-0 hover:bg-emerald-50"
   202→            >
   203→              <span className="min-w-0 truncate">
   204→                <span className="font-medium text-stone-800">{l.customerName}</span>{' '}
   205→                <span className="font-mono text-xs text-stone-400">{l.leadCode}</span>
   206→              </span>
   207→              <span className="shrink-0 text-xs text-stone-500">{l.mobile}</span>
   208→            </button>
   209→          ))
   210→        )}
   211→      </div>
   212→    </div>
   213→  )
   214→}
