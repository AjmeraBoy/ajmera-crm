   453→                        {t.members.length > 5 ? (

   463→                  </div>
   464→                </div>
   465→              ))}
   466→            </div>
   467→          )}
   468→        </TabsContent>
   469→      </Tabs>
   470→
   471→      {/* add / edit user dialog */}

   476→        onSaved={() => setReloadKey((k) => k + 1)}
   477→      />
   478→
   479→      {/* reset password dialog */}

   489→          setReloadKey((k) => k + 1)
   490→          loadTeams()
   491→        }}
   492→      />
   493→    </div>

   499→function UserDialog({
   500→  open,
   501→  onOpenChange,
   502→  user,
   503→  onSaved,
   504→}: {
   505→  open: boolean
   506→  onOpenChange: (open: boolean) => void
   507→  user: UserRow | null
   508→  onSaved: () => void
   509→}) {
   510→  const { toast } = useToast()
   511→  const isEdit = !!user
   512→  const [teams, setTeams] = useState<TeamRow[]>([])
   513→  const [form, setForm] = useState({
   514→    name: '',
   515→    email: '',
   516→    password: '',
   517→    role: 'EXECUTIVE',
   518→    dept: NONE,
   519→    phone: '',
   520→    languages: '',
   521→    teamId: NONE,
   522→    dailyCallTarget: '60',
   523→  })
   524→  const [saving, setSaving] = useState(false)
   525→  const [error, setError] = useState('')
   526→
   527→  useEffect(() => {
   528→    if (!open) return
   529→    setError('')
   530→    setForm(
   531→      user
   532→        ? {
   533→            name: user.name,
   534→            email: user.email,
   535→            password: '',
   536→            role: user.role,
   537→            dept: user.department ?? NONE,
   538→            phone: user.phone ?? '',
   539→            languages: user.languages ?? '',
   540→            teamId: user.teamId ?? NONE,
   541→            dailyCallTarget: String(user.dailyCallTarget ?? 60),
   542→          }
   543→        : { name: '', email: '', password: '', role: 'EXECUTIVE', dept: NONE, phone: '', languages: '', teamId: NONE, dailyCallTarget: '60' }
   544→    )
   545→  }, [open, user])
   546→
   547→  useEffect(() => {
   548→    if (!open) return
   549→    api<{ teams: TeamRow[] }>('/api/teams')
   550→      .then((res) => setTeams(res.teams ?? []))
   551→      .catch(() => setTeams([]))
   552→  }, [open])
   553→
   554→  const set = <K extends keyof typeof form>(key: K, value: (typeof form)[K]) => setForm((f) => ({ ...f, [key]: value }))
   555→
   556→  const submit = async () => {
   557→    setError('')
   558→    if (!form.name.trim() || !form.email.trim() || (!isEdit && !form.password) || !form.role) {
   559→      setError('Name, email, role and password (on create) are required.')
   560→      return
   561→    }
   562→    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(form.email.trim())) {
   563→      setError('Please enter a valid email address.')
   564→      return
   565→    }
   566→    setSaving(true)
   567→    try {
   568→      const body: Record<string, unknown> = {
   569→        name: form.name.trim(),
   570→        email: form.email.trim().toLowerCase(),
   571→        role: form.role,
   572→        department: form.dept === NONE ? null : form.dept,
   573→        phone: form.phone.trim() || null,
   574→        languages: form.languages.trim() || null,
   575→        teamId: form.teamId === NONE ? null : form.teamId,
   576→        dailyCallTarget: Number(form.dailyCallTarget) || 60,
   577→      }
   578→      if (isEdit) {
   579→        if (form.password) body.password = form.password
   580→        await api('/api/users', { method: 'PATCH', body: { id: user.id, ...body } })
   581→        toast({ title: 'User updated', description: `${form.name} saved successfully.` })
   582→      } else {
   583→        body.password = form.password
   584→        await api('/api/users', { method: 'POST', body })
   585→        toast({ title: 'User created', description: `${form.name} can now log in.` })
   586→      }
   587→      onSaved()
   588→      onOpenChange(false)
   589→    } catch (e) {
   590→      setError((e as Error).message)
   591→      toast({ title: isEdit ? 'Update failed' : 'Create failed', description: (e as Error).message, variant: 'destructive' })
   592→    } finally {
   593→      setSaving(false)
   594→    }
   595→  }
   596→
   597→  return (
   598→    <Dialog open={open} onOpenChange={(o) => { if (!saving) onOpenChange(o) }}>
   599→      <DialogContent className="max-h-[92vh] overflow-y-auto sm:max-w-lg">
   600→        <DialogHeader>
   601→          <DialogTitle>{isEdit ? `Edit User — ${user.name}` : 'Add User'}</DialogTitle>
   602→          <DialogDescription>
   603→            {isEdit ? 'Leave password blank to keep the current password.' : 'Create a login for a team member. Share the password securely.'}
   604→          </DialogDescription>
   605→        </DialogHeader>
   606→        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   607→          <div className="space-y-1.5">
   608→            <Label>Name *</Label>
   609→            <Input value={form.name} onChange={(e) => set('name', e.target.value)} disabled={saving} placeholder="Full name" />
   610→          </div>
   611→          <div className="space-y-1.5">
   612→            <Label>Email *</Label>
   613→            <Input type="email" value={form.email} onChange={(e) => set('email', e.target.value)} disabled={saving} placeholder="name@ajmera.com" />
   614→          </div>
   615→          <div className="space-y-1.5">
   616→            <Label>{isEdit ? 'New Password (optional)' : 'Password *'}</Label>
   617→            <Input type="password" value={form.password} onChange={(e) => set('password', e.target.value)} disabled={saving} placeholder={isEdit ? 'Leave blank = unchanged' : 'Min 6 characters'} />
   618→          </div>
   619→          <div className="space-y-1.5">
   620→            <Label>Role *</Label>
   621→            <Select value={form.role} onValueChange={(v) => set('role', v)} disabled={saving}>
   622→              <SelectTrigger aria-label="Role"><SelectValue /></SelectTrigger>
   623→              <SelectContent>
   624→                {ROLES.map((r) => (
   625→                  <SelectItem key={r} value={r}>{ROLE_LABELS[r]}</SelectItem>
   626→                ))}
   627→              </SelectContent>
   628→            </Select>
   629→          </div>
   630→          <div className="space-y-1.5">
   631→            <Label>Department</Label>
   632→            <Select value={form.dept} onValueChange={(v) => set('dept', v)} disabled={saving}>
   633→              <SelectTrigger aria-label="Department"><SelectValue /></SelectTrigger>
   634→              <SelectContent>
   635→                <SelectItem value={NONE}>None</SelectItem>
   636→                {DEPARTMENTS.map((d) => (
   637→                  <SelectItem key={d} value={d}>{DEPT_LABELS[d]}</SelectItem>
   638→                ))}
   639→              </SelectContent>
   640→            </Select>
   641→          </div>
   642→          <div className="space-y-1.5">
   643→            <Label>Team</Label>
   644→            <Select value={form.teamId} onValueChange={(v) => set('teamId', v)} disabled={saving}>
   645→              <SelectTrigger aria-label="Team"><SelectValue /></SelectTrigger>
   646→              <SelectContent>
   647→                <SelectItem value={NONE}>No team</SelectItem>
   648→                {teams.map((t) => (
   649→                  <SelectItem key={t.id} value={t.id}>{t.name} ({DEPT_SHORT[t.department] ?? t.department})</SelectItem>
   650→                ))}
   651→              </SelectContent>
   652→            </Select>
   653→          </div>
   654→          <div className="space-y-1.5">
   655→            <Label>Phone</Label>
   656→            <Input value={form.phone} onChange={(e) => set('phone', e.target.value)} disabled={saving} placeholder="+91…" />
   657→          </div>
   658→          <div className="space-y-1.5">
   659→            <Label>Daily Call Target</Label>
   660→            <Input type="number" min={0} value={form.dailyCallTarget} onChange={(e) => set('dailyCallTarget', e.target.value)} disabled={saving} />
   661→          </div>
   662→          <div className="space-y-1.5 sm:col-span-2">
   663→            <Label>Languages</Label>
   664→            <Input value={form.languages} onChange={(e) => set('languages', e.target.value)} disabled={saving} placeholder="English, Hindi, Gujarati" />
   665→            <p className="text-[11px] text-stone-400">Comma separated list of languages the user speaks.</p>
   666→          </div>
   667→        </div>
   668→        {error ? <p className="rounded-lg border border-rose-200 bg-rose-50 px-3 py-2 text-xs text-rose-700">{error}</p> : null}
   669→        <DialogFooter className="gap-2">
   670→          <Button variant="outline" disabled={saving} onClick={() => onOpenChange(false)}>Cancel</Button>
   671→          <Button className="bg-emerald-600 text-white hover:bg-emerald-700" disabled={saving} onClick={submit}>
   672→            {saving ? <Loader2 className="mr-1 h-4 w-4 animate-spin" aria-hidden /> : null}

   674→          </Button>

   676→      </DialogContent>

   683→function ResetPasswordDialog({ user, onClose }: { user: UserRow | null; onClose: () => void }) {
   684→  const { toast } = useToast()
   685→  const [password, setPassword] = useState('')
   686→  const [saving, setSaving] = useState(false)
   687→  const [error, setError] = useState('')
   688→
   689→  useEffect(() => {
   690→    if (user) {
   691→      setPassword('')
   692→      setError('')
   693→    }
   694→  }, [user])
   695→
   696→  const submit = async () => {
   697→    if (!user) return
   698→    if (password.length < 6) {
   699→      setError('Password must be at least 6 characters.')
   700→      return
   701→    }
   702→    setSaving(true)
   703→    try {
   704→      await api('/api/users', { method: 'PATCH', body: { id: user.id, password } })
   705→      toast({ title: 'Password reset', description: `New password set for ${user.name}.` })
   706→      onClose()
   707→    } catch (e) {
   708→      setError((e as Error).message)
   709→      toast({ title: 'Password reset failed', description: (e as Error).message, variant: 'destructive' })
   710→    } finally {
   711→      setSaving(false)
   712→    }
   713→  }
   714→
   715→  return (
   716→    <Dialog open={!!user} onOpenChange={(o) => { if (!saving && !o) onClose() }}>
   717→      <DialogContent className="sm:max-w-sm">
   718→        <DialogHeader>
   719→          <DialogTitle>Reset Password</DialogTitle>
   720→          <DialogDescription>
   721→            Set a new password for <span className="font-medium text-stone-700">{user?.name}</span> ({user?.email}).
   722→          </DialogDescription>
   723→        </DialogHeader>
   724→        <div className="space-y-1.5">
   725→          <Label>New Password *</Label>
   726→          <Input
   727→            type="password"
   728→            value={password}
   729→            onChange={(e) => setPassword(e.target.value)}
   730→            disabled={saving}
   731→            placeholder="Min 6 characters"
   732→            onKeyDown={(e) => { if (e.key === 'Enter') submit() }}
   733→          />
   734→          {error ? <p className="text-xs text-rose-600">{error}</p> : null}
   735→        </div>
   736→        <DialogFooter className="gap-2">
   737→          <Button variant="outline" disabled={saving} onClick={onClose}>Cancel</Button>
   738→          <Button className="bg-emerald-600 text-white hover:bg-emerald-700" disabled={saving || !password} onClick={submit}>
   739→            {saving ? <Loader2 className="mr-1 h-4 w-4 animate-spin" aria-hidden /> : null}
   740→            Reset Password
   741→          </Button>
   742→        </DialogFooter>
   743→      </DialogContent>
   744→    </Dialog>
   745→  )
   746→}

   748→// ---------- team form dialog ----------

   750→function TeamDialog({
   751→  open,
   752→  onOpenChange,
   753→  team,
   754→  defaultDept,
   755→  onSaved,
   756→}: {
   757→  open: boolean
   758→  onOpenChange: (open: boolean) => void
   759→  team: TeamRow | null
   760→  defaultDept: string
   761→  onSaved: () => void
   762→}) {
   763→  const { toast } = useToast()
   764→  const isEdit = !!team
   765→  const [name, setName] = useState('')
   766→  const [dept, setDept] = useState('ONLINE')
   767→  const [leaderId, setLeaderId] = useState(NONE)
   768→  const [leaders, setLeaders] = useState<{ id: string; name: string; role: string }[]>([])
   769→  const [saving, setSaving] = useState(false)
   770→  const [error, setError] = useState('')
   771→
   772→  useEffect(() => {
   773→    if (!open) return
   774→    setError('')
   775→    setName(team?.name ?? '')
   776→    setDept(team?.department ?? defaultDept ?? 'ONLINE')
   777→    setLeaderId(team?.leader?.id ?? NONE)
   778→  }, [open, team, defaultDept])
   779→
   780→  // load candidate leaders (TEAM_LEADER / MANAGER of the chosen department)
   781→  useEffect(() => {
   782→    if (!open || !dept) {
   783→      setLeaders([])
   784→      return
   785→    }
   786→    let cancelled = false
   787→    api<{ users: { id: string; name: string; role: string }[] }>(`/api/users${qs({ dept, active: 1 })}`)
   788→      .then((res) => {
   789→        if (cancelled) return
   790→        const opts = (res.users ?? []).filter((u) => u.role === 'TEAM_LEADER' || u.role === 'MANAGER')
   791→        if (team?.leader && !opts.some((o) => o.id === team.leader?.id)) {
   792→          opts.unshift({ id: team.leader.id, name: team.leader.name, role: 'TEAM_LEADER' })
   793→        }
   794→        setLeaders(opts)
   795→      })
   796→      .catch(() => { if (!cancelled) setLeaders([]) })
   797→    return () => { cancelled = true }
   798→  }, [open, dept, team])
   799→
   800→  const submit = async () => {
   801→    setError('')
   802→    if (!name.trim()) {
   803→      setError('Team name is required.')
   804→      return
   805→    }
   806→    setSaving(true)
   807→    try {
   808→      if (isEdit) {
   809→        await api('/api/teams', {
   810→          method: 'PATCH',
   811→          body: { id: team.id, name: name.trim(), leaderId: leaderId === NONE ? null : leaderId },
   812→        })
   813→        toast({ title: 'Team updated', description: `${name} saved.` })
   814→      } else {
   815→        await api('/api/teams', {
   816→          method: 'POST',
   817→          body: { name: name.trim(), department: dept, leaderId: leaderId === NONE ? null : leaderId },
   818→        })
   819→        toast({ title: 'Team created', description: `${name} is ready for members.` })
   820→      }
   821→      onSaved()
   822→      onOpenChange(false)
   823→    } catch (e) {
   824→      setError((e as Error).message)
   825→      toast({ title: isEdit ? 'Update failed' : 'Create failed', description: (e as Error).message, variant: 'destructive' })
   826→    } finally {
   827→      setSaving(false)
   828→    }
   829→  }
   830→
   831→  return (
   832→    <Dialog open={open} onOpenChange={(o) => { if (!saving) onOpenChange(o) }}>
   833→      <DialogContent className="sm:max-w-md">
   834→        <DialogHeader>
   835→          <DialogTitle>{isEdit ? `Edit Team — ${team.name}` : 'Add Team'}</DialogTitle>
   836→          <DialogDescription>Teams group executives for assignment, targets and performance views.</DialogDescription>
   837→        </DialogHeader>
   838→        <div className="space-y-3">
   839→          <div className="space-y-1.5">
   840→            <Label>Team Name *</Label>
   841→            <Input value={name} onChange={(e) => setName(e.target.value)} disabled={saving} placeholder="e.g. Export — Team A" />
   842→          </div>
   843→          <div className="space-y-1.5">
   844→            <Label>Department *</Label>
   845→            <Select value={dept} onValueChange={setDept} disabled={saving || isEdit}>
   846→              <SelectTrigger aria-label="Team department"><SelectValue /></SelectTrigger>
   847→              <SelectContent>
   848→                {DEPARTMENTS.map((d) => (
   849→                  <SelectItem key={d} value={d}>{DEPT_LABELS[d]}</SelectItem>
   850→                ))}
   851→              </SelectContent>
   852→            </Select>
   853→            {isEdit ? <p className="text-[11px] text-stone-400">Department cannot be changed after creation.</p> : null}
   854→          </div>
   855→          <div className="space-y-1.5">
   856→            <Label>Team Leader</Label>
   857→            <Select value={leaderId} onValueChange={setLeaderId} disabled={saving}>
   858→              <SelectTrigger aria-label="Team leader"><SelectValue placeholder="Select leader" /></SelectTrigger>
   859→              <SelectContent>
   860→                <SelectItem value={NONE}>No leader</SelectItem>
   861→                {leaders.map((l) => (
   862→                  <SelectItem key={l.id} value={l.id}>{l.name} ({ROLE_LABELS[l.role] ?? l.role})</SelectItem>
   863→                ))}
   864→              </SelectContent>
   865→            </Select>
   866→            <p className="text-[11px] text-stone-400">Team Leaders and Managers of the selected department.</p>
   867→          </div>
   868→          {error ? <p className="rounded-lg border border-rose-200 bg-rose-50 px-3 py-2 text-xs text-rose-700">{error}</p> : null}
   869→        </div>
   870→        <DialogFooter className="gap-2">
   871→          <Button variant="outline" disabled={saving} onClick={() => onOpenChange(false)}>Cancel</Button>
   872→          <Button className="bg-emerald-600 text-white hover:bg-emerald-700" disabled={saving || !name.trim()} onClick={submit}>
   873→            {saving ? <Loader2 className="mr-1 h-4 w-4 animate-spin" aria-hidden /> : null}
   874→            {isEdit ? 'Save Changes' : 'Create Team'}
   875→          </Button>
   876→        </DialogFooter>
   877→      </DialogContent>
   878→    </Dialog>
   879→  )
   880→}
