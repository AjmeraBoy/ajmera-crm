   205→  const [execs, setExecs] = useState<ExecUser[]>([])
   206→  // edit form
   207→  const [formOpen, setFormOpen] = useState(false)
   208→  // stage backward-move confirmation
   209→  const [backStage, setBackStage] = useState<MasterItemDTO | null>(null)
   210→
   211→  // disposition form
   212→  const [dispId, setDispId] = useState('')
   213→  const [subId, setSubId] = useState('')
   214→  const [callbackAt, setCallbackAt] = useState('')
   215→  const [followUpAt, setFollowUpAt] = useState('')
   216→  const [estValue, setEstValue] = useState('')
   217→  const [payAmount, setPayAmount] = useState('')
   218→  const [payMode, setPayMode] = useState('UPI')
   219→  const [dispNote, setDispNote] = useState('')
   220→
   221→  // follow-up schedule (inline) + reschedule dialog
   222→  const [fuDue, setFuDue] = useState('')
   223→  const [fuNote, setFuNote] = useState('')
   224→  const [reschedFollowup, setReschedFollowup] = useState<FollowupRow | null>(null)
   225→  const [reschedTo, setReschedTo] = useState('')
   226→
   227→  // quotation quick create
   228→  const [products, setProducts] = useState<ProductLite[]>([])
   229→  const [productsLoaded, setProductsLoaded] = useState(false)
   230→  const [quoteRows, setQuoteRows] = useState<{ productId: string; qty: string }[]>([{ productId: '', qty: '1' }])
   231→  const [quoteDiscount, setQuoteDiscount] = useState('')
   232→
   233→  // payment dialog
   234→  const [payOpen, setPayOpen] = useState(false)
   235→  const [payInput, setPayInput] = useState('')
   236→  const [payModeInput, setPayModeInput] = useState('BANK_TRANSFER')
   237→  const [payAdvance, setPayAdvance] = useState(false)
   238→  const [payNotes, setPayNotes] = useState('')
   239→
   240→  // ticket dialog
   241→  const [ticketOpen, setTicketOpen] = useState(false)
   242→  const [ticketType, setTicketType] = useState('')
   243→  const [ticketPriority, setTicketPriority] = useState('MEDIUM')
   244→  const [ticketSubject, setTicketSubject] = useState('')
   245→  const [ticketDesc, setTicketDesc] = useState('')
   246→
   247→  // meeting dialog
   248→  const [meetOpen, setMeetOpen] = useState(false)
   249→  const [meetTitle, setMeetTitle] = useState('')
   250→  const [meetAt, setMeetAt] = useState('')
   251→  const [meetLink, setMeetLink] = useState('')
   252→
   253→  // document dialog
   254→  const [docOpen, setDocOpen] = useState(false)
   255→  const [docName, setDocName] = useState('')
   256→  const [docType, setDocType] = useState('OTHER')
   257→  const [docUrl, setDocUrl] = useState('')
   258→
   259→  const lead = bundle?.lead
   260→
   261→  const load = useCallback(async () => {
   262→    if (!id) return
   263→    setLoading(true)
   264→    try {
   265→      const res = await api<LeadBundle>(`/api/leads/detail${qs({ id })}`)
   266→      setBundle(res)
   267→    } catch (e) {
   268→      toast({ title: 'Failed to load lead', description: (e as Error).message, variant: 'destructive' })
   269→    } finally {
   270→      setLoading(false)
   271→    }
   272→  }, [id, toast])
   273→
   274→  useEffect(() => {
   275→    load()
   276→  }, [load, reloadKey])
   277→
   278→  // executives for reassign
   279→  useEffect(() => {
   280→    if (!isManagement) return
   281→    api<{ users: ExecUser[] }>(`/api/users${qs({ role: 'EXECUTIVE', active: 1, dept: lead?.department || undefined })}`)
   282→      .then((res) => setExecs(res.users ?? []))
   283→      .catch(() => setExecs([]))
   284→  }, [isManagement, lead?.department])
   285→
   286→  // prefill disposition form whenever lead changes
   287→  useEffect(() => {
   288→    if (!lead) return
   289→    setDispId(lead.dispositionId ?? '')
   290→    setSubId(lead.subDispositionId ?? '')
   291→    setCallbackAt('')
   292→    setFollowUpAt('')
   293→    setEstValue('')
   294→    setPayAmount('')
   295→    setPayMode('UPI')
   296→    setDispNote('')
   297→  }, [lead?.id, lead?.dispositionId, lead?.subDispositionId, lead])
   298→
   299→  // lazy-load products when quotations tab opens
   300→  useEffect(() => {
   301→    if (tab !== 'quotations' || productsLoaded || !id) return
   302→    setProductsLoaded(true)
   303→    api<{ products: ProductLite[] }>(`/api/products${qs({ active: 1, pageSize: 100 })}`)
   304→      .then((res) => setProducts(res.products ?? []))
   305→      .catch(() => toast({ title: 'Failed to load products', variant: 'destructive' }))
   306→  }, [tab, productsLoaded, id, toast])
   307→
   308→  const dept = lead?.department ?? ''
   309→  const stages = useMemo(() => masters.items('pipeline_stage', dept || undefined), [masters, dept])
   310→  const dispositionOptions = useMemo(() => masters.items('disposition', dept || undefined), [masters, dept])
   311→  const subOptions = useMemo(() => (dispId ? masters.subItems(dispId, dept || undefined) : []), [masters, dispId, dept])
   312→  const dispExtra = useMemo(() => dispositionExtra(masters.byId(dispId)?.extra), [masters, dispId])
   313→  const ticketTypeOptions = useMemo(() => masters.items('ticket_type').map((m) => ({ id: m.id, label: m.label })), [masters])
   314→
   315→  const isOverdue = (d: string | null) => Boolean(d && new Date(d).getTime() < Date.now())
   316→
   317→  // ---------- mutations ----------
   318→
   319→  const runMutation = async (key: string, fn: () => Promise<LeadBundle | void>, success: string) => {
   320→    setPending(key)
   321→    try {
   322→      const res = await fn()
   323→      if (res && typeof res === 'object' && 'lead' in res) setBundle(res as LeadBundle)
   324→      else setReloadKey((k) => k + 1)
   325→      toast({ title: success })
   326→      return true
   327→    } catch (e) {
   328→      toast({ title: 'Action failed', description: (e as Error).message, variant: 'destructive' })
   329→      return false
   330→    } finally {
   331→      setPending(null)
   332→    }
   333→  }
   334→
   335→  const moveStage = async (stage: MasterItemDTO) => {
   336→    if (!lead || stage.id === lead.stageId) return
   337→    const currentIdx = stages.findIndex((s) => s.id === lead.stageId)
   338→    const nextIdx = stages.findIndex((s) => s.id === stage.id)
   339→    if (currentIdx >= 0 && nextIdx >= 0 && nextIdx < currentIdx) {
   340→      setBackStage(stage)
   341→      return
   342→    }
   343→    await doStageMove(stage)
   344→  }
   345→
   346→  const doStageMove = async (stage: MasterItemDTO) => {
   347→    if (!lead) return
   348→    await runMutation(
   349→      'stage',
   350→      () => api<LeadBundle>('/api/leads', { method: 'PATCH', body: { id: lead.id, stageId: stage.id } }),
   351→      `Stage updated — ${stage.label}`
   352→    )
   353→  }
   354→
   355→  const submitDisposition = async () => {
   356→    if (!lead) return
   357→    if (!dispId) {
   358→      toast({ title: 'Select a disposition first', variant: 'destructive' })
   359→      return
   360→    }
   361→    if (dispExtra.showCallback && !callbackAt) {
   362→      toast({ title: 'Next Callback Date & Time is required', variant: 'destructive' })
   363→      return
   364→    }
   365→    if (dispExtra.showFollowUp && !followUpAt) {
   366→      toast({ title: 'Follow-up Date & Time is required', variant: 'destructive' })
   367→      return
   368→    }
   369→    await runMutation(
   370→      'disposition',
   371→      () =>
   372→        api<LeadBundle>('/api/leads', {
   373→          method: 'PATCH',
   374→          body: {
   375→            id: lead.id,
   376→            dispositionUpdate: {
   377→              dispositionId: dispId,
   378→              subDispositionId: subId || undefined,
   379→              callbackAt: dispExtra.showCallback ? callbackAt : undefined,
   380→              followUpAt: dispExtra.showFollowUp ? followUpAt : undefined,
   381→              estimatedValue: dispExtra.showEstimated && estValue ? Number(estValue) : undefined,
   382→              paymentAmount: dispExtra.showPayment && payAmount ? Number(payAmount) : undefined,
   383→              mode: dispExtra.showPayment ? payMode : undefined,
   384→              note: dispNote || undefined,
   385→            },
   386→          },
   387→        }),
   388→      'Disposition updated'
   389→    )

   398→      'note',
   399→      () => api('/api/leads/note', { method: 'POST', body: { leadId: lead.id, note: noteText.trim() } }),
   400→      'Note added'
   401→    )

   403→      setNoteText('')
   404→      setNoteOpen(false)
   405→    }
   406→  }
   407→

   409→    if (!lead || !reassignTarget) {
   410→      toast({ title: 'Select an executive', variant: 'destructive' })
   411→      return
   412→    }
   413→    const okDone = await runMutation(
   414→      'reassign',
   415→      () => api('/api/leads/assign', { method: 'POST', body: { leadIds: [lead.id], assignedToId: reassignTarget } }),
   416→      'Lead reassigned'
   417→    )
   418→    if (okDone) setReassignOpen(false)
   419→  }
   420→
   421→  const scheduleFollowup = async () => {
   422→    if (!lead) return
   423→    if (!fuDue) {
   424→      toast({ title: 'Pick a date & time', variant: 'destructive' })
   425→      return
   426→    }
   427→    const okDone = await runMutation(
   428→      'followup',
   429→      () => api('/api/followups', { method: 'POST', body: { leadId: lead.id, dueAt: new Date(fuDue).toISOString(), note: fuNote || undefined } }),
   430→      'Follow-up scheduled'
   431→    )

   434→      setFuNote('')

   440→      'complete',
   441→      () => api('/api/followups', { method: 'PATCH', body: { id: fu.id, status: 'COMPLETED', outcome: outcome ?? fu.outcome ?? undefined } }),
   442→      'Follow-up completed'
   443→    )
   444→  }
   445→

   447→    if (!reschedFollowup || !reschedTo) {
   448→      toast({ title: 'Pick a new date & time', variant: 'destructive' })
   449→      return
   450→    }

   452→      'resched',
   453→      () =>
   454→        api('/api/followups', {
   455→          method: 'PATCH',
   456→          body: { id: reschedFollowup.id, status: 'RESCHEDULED', rescheduleTo: new Date(reschedTo).toISOString() },

   458→      'Follow-up rescheduled'
   459→    )
   460→    if (okDone) setReschedFollowup(null)
   461→  }

   466→      'call',
   467→      () => api('/api/calls', { method: 'POST', body: { leadId: lead.id, direction: 'INCOMING', status: 'CONNECTED', durationSec: 0 } }),
   468→      'Incoming call logged'
   469→    )
   470→  }

   475→      .filter((r) => r.productId)
   476→      .map((r) => {

   479→      })

   481→      toast({ title: 'Add at least one product', variant: 'destructive' })
   482→      return
   483→    }
   484→    const okDone = await runMutation(
   485→      'quote',
   486→      () => api('/api/quotations', { method: 'POST', body: { leadId: lead.id, items, discount: quoteDiscount ? Number(quoteDiscount) : undefined } }),
   487→      'Quotation created'
   488→    )
   489→    if (okDone) {
   490→      setQuoteRows([{ productId: '', qty: '1' }])
   491→      setQuoteDiscount('')
   492→    }
   493→  }
   494→
   495→  const submitPayment = async () => {
   496→    if (!lead) return
   497→    if (!payInput || Number(payInput) <= 0) {
   498→      toast({ title: 'Enter a valid amount', variant: 'destructive' })
   499→      return
   500→    }
   501→    const okDone = await runMutation(
   502→      'pay',
   503→      () =>
   504→        api('/api/payments', {
   505→          method: 'POST',
   506→          body: {
   507→            leadId: lead.id,
   508→            amount: Math.round(Number(payInput)),
   509→            mode: payModeInput,
   510→            isAdvance: payAdvance,
   511→            notes: payNotes || undefined,
   512→          },
   513→        }),
   514→      'Payment recorded'
   515→    )
   516→    if (okDone) {
   517→      setPayOpen(false)
   518→      setPayInput('')
   519→      setPayNotes('')
   520→      setPayAdvance(false)
   521→    }
   522→  }
   523→
   524→  const submitTicket = async () => {
   525→    if (!lead) return
   526→    if (!ticketSubject.trim()) {
   527→      toast({ title: 'Subject is required', variant: 'destructive' })
   528→      return
   529→    }
   530→    const okDone = await runMutation(
   531→      'ticket',
   532→      () =>
