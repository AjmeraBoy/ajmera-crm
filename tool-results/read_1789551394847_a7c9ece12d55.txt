   155→const MANAGEMENT_ROLES = ['TEAM_LEADER', 'MANAGER', 'ADMIN', 'SUPER_ADMIN']
   156→const PAGE_SIZE = 20
   157→
   158→function defaultFilters(dept: string): Filters {
   159→  return {
   160→    q: '', dept, stageId: '', dispositionId: '', sourceId: '', priority: '', status: '',
   161→    sticky: false, assignedToId: '', from: '', to: '', followup: '',
   162→  }

   166→
   167→export default function LeadsView() {
   168→  const user = useAppStore((s) => s.user)
   169→  const setView = useAppStore((s) => s.setView)
   170→  const { toast } = useToast()
   171→  const masters = useMasters()
   172→
   173→  const isManagement = MANAGEMENT_ROLES.includes(user?.role ?? '')
   174→  const isAdmin = user?.role === 'ADMIN' || user?.role === 'SUPER_ADMIN'
   175→  const canPickDept = !user?.department
   176→
   177→  const [filters, setFilters] = useState<Filters>(() => defaultFilters(user?.department ?? ''))
   178→  const [qInput, setQInput] = useState('')
   179→  const [page, setPage] = useState(1)
   180→  const [data, setData] = useState<LeadsResponse | null>(null)
   181→  const [loading, setLoading] = useState(true)
   182→  const [reloadKey, setReloadKey] = useState(0)
   183→
   184→  const [execs, setExecs] = useState<ExecUser[]>([])
   185→  const [selected, setSelected] = useState<Set<string>>(new Set())
   186→
   187→  const [formOpen, setFormOpen] = useState(false)
   188→  const [editLead, setEditLead] = useState<LeadRow | null>(null)
   189→  const [assignOpen, setAssignOpen] = useState(false)
   190→  const [assignIds, setAssignIds] = useState<string[]>([])
   191→  const [assignTarget, setAssignTarget] = useState('')
   192→  const [assigning, setAssigning] = useState(false)
   193→  const [importOpen, setImportOpen] = useState(false)
   194→  const [deleteLead, setDeleteLead] = useState<LeadRow | null>(null)
   195→  const [deleting, setDeleting] = useState(false)
   196→  const [exporting, setExporting] = useState(false)
   197→
   198→  // debounce search input
   199→  useEffect(() => {
   200→    const t = setTimeout(() => {
   201→      setFilters((f) => (f.q === qInput ? f : { ...f, q: qInput }))
   202→      setPage(1)
   203→    }, 400)
   204→    return () => clearTimeout(t)
   205→  }, [qInput])
   206→
   207→  const buildParams = useCallback(
   208→    (forExport = false) => ({
   209→      q: filters.q || undefined,
   210→      dept: filters.dept || undefined,
   211→      stageId: filters.stageId || undefined,
   212→      dispositionId: filters.dispositionId || undefined,
   213→      sourceId: filters.sourceId || undefined,
   214→      priority: filters.priority || undefined,
   215→      status: filters.status || undefined,
   216→      sticky: filters.sticky ? 1 : undefined,
   217→      assignedToId: filters.assignedToId || undefined,
   218→      from: filters.from || undefined,
   219→      to: filters.to || undefined,
   220→      followup: filters.followup || undefined,
   221→      ...(forExport ? {} : { page, pageSize: PAGE_SIZE }),
   222→    }),
   223→    [filters, page]
   224→  )
   225→
   226→  const load = useCallback(async () => {
   227→    setLoading(true)
   228→    try {
   229→      const res = await api<LeadsResponse>(`/api/leads${qs(buildParams())}`)
   230→      setData(res)
   231→    } catch (e) {
   232→      toast({ title: 'Failed to load leads', description: (e as Error).message, variant: 'destructive' })
   233→    } finally {
   234→      setLoading(false)
   235→    }
   236→  }, [buildParams, toast])
   237→
   238→  useEffect(() => {
   239→    load()
   240→  }, [load, reloadKey])
   241→
   242→  // executives for "Assigned to" filter + assignment dialogs
   243→  useEffect(() => {
   244→    const dept = user?.department ?? filters.dept
   245→    api<{ users: ExecUser[] }>(`/api/users${qs({ role: 'EXECUTIVE', active: 1, dept: dept || undefined })}`)
   246→      .then((res) => setExecs(res.users ?? []))
   247→      .catch(() => setExecs([]))
   248→  }, [user?.department, filters.dept])
   249→
   250→  const execOptions = useMemo<MasterOption[]>(() => execs.map((u) => ({ id: u.id, label: u.name })), [execs])
   251→
   252→  const setFilter = <K extends keyof Filters>(key: K, value: Filters[K]) => {
   253→    setFilters((f) => ({ ...f, [key]: value }))
   254→    setPage(1)
   255→    setSelected(new Set())
   256→  }
   257→
   258→  const resetFilters = () => {
   259→    setFilters(defaultFilters(user?.department ?? ''))
   260→    setQInput('')
   261→    setPage(1)
   262→    setSelected(new Set())
   263→  }
   264→
   265→  // ---------- summary chips ----------
   266→  const summary = data?.summary
   267→  const chips: { key: string; label: string; count: number; active: boolean; onClick: () => void }[] = [
   268→    {
   269→      key: 'all', label: 'Total', count: summary?.total ?? 0,
   270→      active: !filters.status && !filters.sticky,
   271→      onClick: () => {
   272→        setFilters((f) => ({ ...f, status: '', sticky: false }))
   273→        setPage(1)
   274→        setSelected(new Set())
   275→      },
   276→    },
   277→    {
   278→      key: 'active', label: 'Active', count: summary?.active ?? 0,
   279→      active: filters.status === 'ACTIVE' && !filters.sticky,
   280→      onClick: () => setFilter('status', 'ACTIVE'),
   281→    },
   282→    {
   283→      key: 'converted', label: 'Converted', count: summary?.converted ?? 0,
   284→      active: filters.status === 'CONVERTED',
   285→      onClick: () => setFilter('status', 'CONVERTED'),
   286→    },
   287→    {
   288→      key: 'lost', label: 'Lost', count: summary?.lost ?? 0,
   289→      active: filters.status === 'LOST',
   290→      onClick: () => setFilter('status', 'LOST'),
   291→    },
   292→    {
   293→      key: 'sticky', label: 'Sticky', count: summary?.sticky ?? 0,
   294→      active: filters.sticky,
   295→      onClick: () => {
   296→        setFilters((f) => ({ ...f, sticky: !f.sticky, status: '' }))
   297→        setPage(1)
   298→      },
   299→    },
   300→  ]
   301→
   302→  // ---------- actions ----------
   303→  const openLead = (lead: LeadRow) => setView('lead-detail', { leadId: lead.id })
   304→
   305→  const exportCsv = async () => {
   306→    setExporting(true)
   307→    try {
   308→      const res = await api<{ rows: ExportRow[] }>(`/api/leads/export${qs(buildParams(true))}`)
   309→      const rows = (res.rows ?? []).map((r) => ({
   310→        ...r,
   311→        estimatedValue: r.estimatedValue ?? 0,
   312→        nextFollowUpAt: r.nextFollowUpAt ? formatDateTime(r.nextFollowUpAt) : '',
   313→        createdAt: r.createdAt ? formatDateTime(r.createdAt) : '',
   314→      }))
   315→      downloadCSV(`leads-${new Date().toISOString().slice(0, 10)}.csv`, rows)
   316→      toast({ title: 'Export ready', description: `${rows.length} leads exported to CSV` })
   317→    } catch (e) {
   318→      toast({ title: 'Export failed', description: (e as Error).message, variant: 'destructive' })
   319→    } finally {
   320→      setExporting(false)
   321→    }
   322→  }
   323→
   324→  const submitAssign = async (ids: string[]) => {
   325→    if (ids.length === 0) return
   326→    if (!assignTarget) {
   327→      toast({ title: 'Select an executive', variant: 'destructive' })
   328→      return
   329→    }
   330→    setAssigning(true)
   331→    try {
   332→      const res = await api<{ updated: number }>('/api/leads/assign', {
   333→        method: 'POST',
   334→        body: { leadIds: ids, assignedToId: assignTarget },
   335→      })
   336→      toast({ title: 'Leads reassigned', description: `${res.updated} lead(s) assigned successfully` })
   337→      setAssignOpen(false)
   338→      setAssignTarget('')
   339→      setSelected(new Set())
   340→      setReloadKey((k) => k + 1)
   341→    } catch (e) {
   342→      toast({ title: 'Reassign failed', description: (e as Error).message, variant: 'destructive' })
   343→    } finally {
   344→      setAssigning(false)
   345→    }
   346→  }
   347→
   348→  const confirmDelete = async () => {
   349→    if (!deleteLead) return
   350→    setDeleting(true)
   351→    try {
   352→      await api(`/api/leads${qs({ id: deleteLead.id })}`, { method: 'DELETE' })
   353→      toast({ title: 'Lead deleted', description: `${deleteLead.leadCode} removed` })
   354→      setDeleteLead(null)
   355→      setReloadKey((k) => k + 1)
   356→    } catch (e) {
   357→      toast({ title: 'Delete failed', description: (e as Error).message, variant: 'destructive' })
   358→    } finally {
   359→      setDeleting(false)
   360→    }
   361→  }
   362→
   363→  // ---------- bulk selection ----------
   364→  const rowIds = (data?.leads ?? []).map((l) => l.id)
   365→  const allSelected = rowIds.length > 0 && rowIds.every((id) => selected.has(id))
   366→  const toggleAll = () => {
   367→    setSelected(allSelected ? new Set() : new Set(rowIds))
   368→  }
   369→  const toggleOne = (id: string) => {
   370→    setSelected((prev) => {
   371→      const next = new Set(prev)
   372→      if (next.has(id)) next.delete(id)
   373→      else next.add(id)
   374→      return next
   375→    })
   376→  }
   377→
   378→  // ---------- table columns ----------
   379→  const stageOptions = useMemo<MasterOption[]>(
   380→    () => masters.items('pipeline_stage', filters.dept || user?.department || undefined).map((m) => ({ id: m.id, label: m.label })),
   381→    [masters, filters.dept, user?.department]
   382→  )
   383→  const dispositionOptions = useMemo<MasterOption[]>(
   384→    () => masters.items('disposition', filters.dept || user?.department || undefined).map((m) => ({ id: m.id, label: m.label })),
   385→    [masters, filters.dept, user?.department]
   386→  )
   387→  const sourceOptions = useMemo<MasterOption[]>(
   388→    () => masters.items('lead_source', filters.dept || user?.department || undefined).map((m) => ({ id: m.id, label: m.label })),
   389→    [masters, filters.dept, user?.department]
   390→  )
   391→
   392→  const isOverdue = (d: string | null) => Boolean(d && new Date(d).getTime() < Date.now())
   393→
   394→  const columns: Column<LeadRow>[] = useMemo(() => {
   395→    const cols: Column<LeadRow>[] = []
   396→    if (isManagement) {
   397→      cols.push({
   398→        key: 'sel',
   399→        header: '',
   400→        className: 'w-10',
   401→        render: (row) => (
   402→          <div onClick={(e) => e.stopPropagation()}>
   403→            <Checkbox
   404→              checked={selected.has(row.id)}
   405→              onCheckedChange={() => toggleOne(row.id)}
   406→              aria-label={`Select lead ${row.leadCode}`}
   407→            />
   408→          </div>
   409→        ),
   410→      })
   411→    }
   412→    cols.push(
   413→      {
   414→        key: 'leadCode',
   415→        header: 'Lead ID',
   416→        render: (row) => (
   417→          <div className="flex items-center gap-1.5">
   418→            <span className="font-mono text-xs font-bold text-stone-800">{row.leadCode}</span>
   419→            {row.isSticky ? <Pin className="h-3 w-3 text-amber-500" aria-label="Sticky lead" /> : null}
   420→            <StatusBadge status={row.department} variant="dept" className="ml-1 hidden md:inline-flex" />
   421→          </div>
   422→        ),
   423→      },
   424→      {
   425→        key: 'customerName',
   426→        header: 'Customer',
   427→        render: (row) => (
   428→          <div className="min-w-[140px]">
   429→            <p className="font-medium text-stone-800">{row.customerName}</p>
   430→            {row.companyName ? <p className="text-xs text-stone-500">{row.companyName}</p> : null}
   431→            {row.city ? <p className="text-xs text-stone-400">{row.city}{row.state ? `, ${row.state.label}` : row.country ? `, ${row.country.label}` : ''}</p> : null}
   432→          </div>
   433→        ),
   434→      },
   435→      {
   436→        key: 'mobile',
   437→        header: 'Contact',
   438→        render: (row) => (
   439→          <div className="min-w-[120px]">
   440→            <p className="text-sm">{row.mobile}</p>
   441→            {row.whatsapp && row.whatsapp !== row.mobile ? <p className="text-xs text-stone-500">WA: {row.whatsapp}</p> : null}
   442→            {row.email ? <p className="truncate text-xs text-stone-400">{row.email}</p> : null}
   443→          </div>
   444→        ),
   445→      },
   446→      {
   447→        key: 'stage',
   448→        header: 'Stage',
   449→        render: (row) => {
   450→          const color = String(masterExtra(row.stage?.extra).color ?? '')
   451→          return row.stage ? (
   452→            <Badge variant="outline" className="gap-1.5 whitespace-nowrap border-stone-200 font-medium text-stone-700">
   453→              <span className="h-2 w-2 rounded-full" style={color ? { backgroundColor: color } : { backgroundColor: '#a8a29e' }} aria-hidden />
   454→              {row.stage.label}
   455→            </Badge>
   456→          ) : (
   457→            <span className="text-stone-400">—</span>
   458→          )
   459→        },
   460→      },
   461→      {
   462→        key: 'disposition',
   463→        header: 'Disposition',
   464→        render: (row) => (
   465→          <div className="min-w-[110px]">
   466→            <p className="whitespace-nowrap">{row.disposition?.label ?? <span className="text-stone-400">—</span>}</p>
   467→            {row.subDisposition ? <p className="text-xs text-stone-400">{row.subDisposition.label}</p> : null}
   468→          </div>
   469→        ),
   470→      },
   471→      {
   472→        key: 'priority',
   473→        header: 'Priority',
   474→        render: (row) => <StatusBadge status={row.priority} variant="priority" />,
   475→      },
   476→      {
   477→        key: 'estimatedValue',
   478→        header: 'Value',
   479→        className: 'text-right',
   480→        render: (row) => (
   481→          <span className="font-medium text-emerald-700">{row.estimatedValue ? formatINR(row.estimatedValue) : '—'}</span>
   482→        ),
   483→      },
   484→      {
   485→        key: 'owner',
   486→        header: 'Owner',
   487→        render: (row) =>
   488→          row.assignedTo ? (
   489→            <div className="flex items-center gap-2">
   490→              <UserAvatar name={row.assignedTo.name} className="h-6 w-6" />
   491→              <span className="whitespace-nowrap text-xs">{row.assignedTo.name}</span>
   492→            </div>
   493→          ) : (
   494→            <span className="text-stone-400">Unassigned</span>
   495→          ),
   496→      },
   497→      {
   498→        key: 'nextFollowUpAt',
   499→        header: 'Next Follow-up',
   500→        render: (row) =>
   501→          row.nextFollowUpAt ? (
   502→            <span className={cn('whitespace-nowrap text-xs', isOverdue(row.nextFollowUpAt) && 'font-semibold text-rose-600')}>
   503→              {formatDateTime(row.nextFollowUpAt)}
   504→            </span>
   505→          ) : (
   506→            <span className="text-stone-400">—</span>
   507→          ),
   508→      },
   509→      {
   510→        key: 'status',
   511→        header: 'Status',
   512→        render: (row) => <StatusBadge status={row.status} variant="leadStatus" />,
   513→      },
   514→      {
   515→        key: 'actions',
   516→        header: '',
   517→        className: 'w-12',
   518→        render: (row) => (
   519→          <div onClick={(e) => e.stopPropagation()}>
   520→            <DropdownMenu>
   521→              <DropdownMenuTrigger asChild>
   522→                <Button variant="ghost" size="icon" className="h-8 w-8" aria-label={`Actions for ${row.leadCode}`}>
   523→                  <MoreHorizontal className="h-4 w-4" aria-hidden />
   524→                </Button>
   525→              </DropdownMenuTrigger>
   526→              <DropdownMenuContent align="end">
   527→                <DropdownMenuItem onClick={() => openLead(row)}>Open 360° view</DropdownMenuItem>
   528→                <DropdownMenuItem onClick={() => setView('dialer', { leadId: row.id })}>
   529→                  <Phone className="mr-2 h-4 w-4" aria-hidden /> Call
   530→                </DropdownMenuItem>
   531→                <DropdownMenuItem onClick={() => setView('whatsapp', { leadId: row.id })}>
   532→                  <MessageCircle className="mr-2 h-4 w-4" aria-hidden /> WhatsApp
   533→                </DropdownMenuItem>
   534→                {isManagement ? (
   535→                  <DropdownMenuItem
   536→                    onClick={() => {
   537→                      setAssignIds([row.id])
   538→                      setAssignTarget(row.assignedTo?.id ?? '')
   539→                      setAssignOpen(true)
   540→                    }}
   541→                  >
   542→                    <UserCog className="mr-2 h-4 w-4" aria-hidden /> Assign / Reassign
   543→                  </DropdownMenuItem>
   544→                ) : null}
   545→                {isAdmin ? (
   546→                  <>
   547→                    <DropdownMenuSeparator />
   548→                    <DropdownMenuItem className="text-rose-600 focus:text-rose-700" onClick={() => setDeleteLead(row)}>
   549→                      <Trash2 className="mr-2 h-4 w-4" aria-hidden /> Delete
   550→                    </DropdownMenuItem>

   882→const HEADER_ALIASES: Record<string, string> = {
   883→  customername: 'customerName', name: 'customerName', customer: 'customerName',
   884→  mobile: 'mobile', phone: 'mobile', number: 'mobile',
   885→  whatsapp: 'whatsapp', email: 'email', city: 'city',
   886→  company: 'companyName', companyname: 'companyName',

   891→function parseImportCsv(text: string): { rows: Record<string, string>[]; error: string | null } {
   892→  const lines = text.split(/\r?\n/).map((l) => l.trim()).filter(Boolean)
   893→  if (lines.length === 0) return { rows: [], error: 'Paste at least one data row' }
   894→  let headers: string[] | null = null
   895→  const first = lines[0].toLowerCase().replace(/[\s_]/g, '')
   896→  if (first.includes('customer') || first.includes('mobile') || first.includes('name')) {
   897→    headers = lines[0].split(',').map((h) => HEADER_ALIASES[h.trim().toLowerCase().replace(/[\s_]/g, '')] ?? '')
   898→    lines.shift()
   899→  }
   900→  const positional = ['customerName', 'mobile', 'email', 'city', 'productInterest', 'notes']
   901→  const rows = lines.map((line) => {
   902→    const cols = line.split(',').map((c) => c.trim())
   903→    const row: Record<string, string> = {}
   904→    if (headers) {
   905→      headers.forEach((h, i) => {
   906→        if (h && cols[i] !== undefined && cols[i] !== '') row[h] = cols[i]
   907→      })
   908→    } else {
   909→      positional.forEach((h, i) => {
   910→        if (cols[i] !== undefined && cols[i] !== '') row[h] = cols[i]
   911→      })
   912→    }
   913→    return row
   914→  })
   915→  return { rows, error: null }
   916→}
